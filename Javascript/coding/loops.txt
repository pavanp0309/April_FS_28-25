### 🧮 1. Reverse a Number (Without using toString)
```js
function reverseNumber(num) {
  let rev = 0;
  for (; num > 0; num = Math.floor(num / 10)) {
    let digit = num % 10;
    rev = rev * 10 + digit;
  }
  return rev;
}
```

### 🧮 2. Reverse a Number (Using String Method)
```js
function reverseNumberStr(num) {
  let str = num.toString();
  let rev = '';
  for (let i = str.length - 1; i >= 0; i--) {
    rev += str[i];
  }
  return parseInt(rev);
}
```

### 🔤 3. Count Vowels in a String (Using for loop)
```js
function countVowels(str) {
  let count = 0;
  const vowels = 'aeiouAEIOU';
  for (let i = 0; i < str.length; i++) {
    if (vowels.includes(str[i])) count++;
  }
  return count;
}
```

### 🔤 4. Count Consonants in a String (Using for loop)
```js
function countConsonants(str) {
  let count = 0;
  const vowels = 'aeiouAEIOU';
  for (let i = 0; i < str.length; i++) {
    let ch = str[i].toLowerCase();
    if (/[a-z]/.test(ch) && !vowels.includes(ch)) {
      count++;
    }
  }
  return count;
}
```

### 🧠 7. Count Even Digits in a Number (Without string)
```js
function countEvenDigits(num) {
  let count = 0;
  for (; num > 0; num = Math.floor(num / 10)) {
    let digit = num % 10;
    if (digit % 2 === 0) count++;
  }
  return count;
}
```

### 🧠 8. Count Even Digits (With string method)
```js
function countEvenDigitsStr(num) {
  let str = num.toString();
  let count = 0;
  for (let i = 0; i < str.length; i++) {
    if (parseInt(str[i]) % 2 === 0) count++;
  }
  return count;
}
```
