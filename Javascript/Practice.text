function oddDiv5(num){
    if(num%2!==0){ // checks given number is odd
        if(num%5==0){ //checks given number is div by 5
            console.log("given number is Odd and divisible by 5")
        }else{
            console.log("given number is odd not divisible 5")
        }
    }else{
